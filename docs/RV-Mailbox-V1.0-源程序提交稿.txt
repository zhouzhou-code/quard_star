  126  /* 把 MMIO 偏移解码成 (bank, channel, 通道内寄存器)；返回 false 表示非通道区 */
  127  static bool qsmb_decode(hwaddr offset, int *bank, int *ch, hwaddr *reg)
  128  {
  129      hwaddr o;
  130      if (offset < RVMB_BANK_TL_BASE + RVMB_BANK_SPAN) {   /* TL_BASE=0 */
  131          *bank = RVMB_BANK_TO_LINUX;
  132          o = offset - RVMB_BANK_TL_BASE;
  133      } else if (offset >= RVMB_BANK_TR_BASE && offset < RVMB_BANK_TR_BASE + RVMB_BANK_SPAN) {
  134          *bank = RVMB_BANK_TO_RTOS;
  135          o = offset - RVMB_BANK_TR_BASE;
  136      } else {
  137          return false;
  138      }
  139      *ch = o / RVMB_CH_STRIDE;
  140      *reg = o % RVMB_CH_STRIDE;
  141      return true;
  142  }

  157                                          unsigned size)
  158  {
  159      RISCVMailboxState *s = RISCV_MAILBOX(opaque);
  160      int bank, ch;
  161      hwaddr reg;

  163      switch (offset) {
  164      case R_REVISION:

  166      case R_NUM_CHANNELS:

  273  static void riscv_mailbox_reset(DeviceState *dev)
  274  {
  275      RISCVMailboxState *s = RISCV_MAILBOX(dev);
  276      int b, ch;
  277  
  278      /* 复位：状态全 0；屏蔽全 1（默认全部屏蔽，接收方需显式 unmask）*/

  280          for (ch = 0; ch < RVMB_NUM_CHANNELS; ch++) {
  281              s->stat[b][ch] = 0;
  282              s->mask[b][ch] = 0xFFFFFFFF;
  283          }
  284      }
  285      qemu_set_irq(s->irq_linux, 0);
  286      qemu_set_irq(s->irq_rtos, 0);

  375  static irqreturn_t rvmb_isr(int irq, void *dev_id)
  376  {
  377  	struct rvmb_selftest *st = dev_id;
  378  
  379  	/* 读门铃状态 */
  380  	st->isr_stat = readl(st->base + RVMB_STAT);
  381  	/* W1C 清门铃位，电平随之落下 */
  382  	writel(RVMB_DBELL_BIT, st->base + RVMB_CLEAR);
  383  	st->irq_count++;
  384  
  385  	complete(&st->done);
  386  	return IRQ_HANDLED;
  387  }
